    private boolean architectureTarget(MinecraftClient c){
        c.gameRenderer.updateTargetedEntity(1);
        if(!(c.crosshairTarget instanceof BlockHitResult actual)||actual.getType()!=HitResult.Type.BLOCK)return false;
        var owner=CompositeRuntime.targetReadOnly(c.world,actual.getBlockPos(),c.player);
        if(owner!=null&&owner.instanceId().equals(current))return true;
        // Unshifted native walls/ladders retain vanilla outline and attack paths.
        // Their resident BE has an identity before any mounted contribution exists.
        if(!actual.getBlockPos().equals(pad)||!VerticalMount.isNative(c.world.getBlockState(pad))||VerticalMount.offset(c.world,pad)!=0)return false;
        var resident=VerticalMount.owner(c.world,pad);
        if(resident==null||!resident.instanceId().equals(current))return false;
        JsonObject route=new JsonObject();route.addProperty("path","ACTUAL_UNSHIFTED_NATIVE_BLOCK_HIT");route.add("blockHit",hit(actual));route.addProperty("residentUuid",resident.instanceId().toString());route.addProperty("verticalOffset",0);
        caseResult.add("actualNativeAttackTarget",route);
        return true;
    }
