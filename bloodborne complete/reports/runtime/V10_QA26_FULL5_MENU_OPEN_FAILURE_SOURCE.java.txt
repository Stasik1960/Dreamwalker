 public static boolean tick(MinecraftClient client){
  if(done)return true;
  try{
   if(client.world==null||client.player==null||client.interactionManager==null||client.getServer()==null)return false;
   if(mouseRelease){client.options.attackKey.setPressed(false);mouseRelease=false;}
   if(!initialized)initialize(client);
   if(prepFailure!=null)throw new IllegalStateException("Marker preparation failed",prepFailure);
   if(++total>12000)throw new IllegalStateException("Bounded menus QA exceeded600seconds");
   if(index==PLAN.size()){done=true;RESULT.addProperty("status","PASS_ACTUAL_CLIENT_MENUS_PACKETS_RULES_COMMANDS");RESULT.addProperty("completedSteps",STEPS.size());if(mode.equals("REENTER"))RESULT.addProperty("ordinaryClientReentryActual","PASS_DISTINCT_SAVED_WORLD_GUI");client.options.getGuiScale().setValue(originalGuiScale);client.onResolutionChanged();return true;}
   Step step=PLAN.get(index);age++;
   if(!started){step.action().run(client);started=true;return false;}
   if(age>=3&&step.check().test(client)){
    JsonObject row=new JsonObject();row.addProperty("name",step.name());row.addProperty("status","PASS");row.addProperty("clientTicks",age);STEPS.add(row);index++;age=0;started=false;
   }else if(age>step.limit())throw new IllegalStateException("GUI step timed out: "+step.name()+" screen="+screenName(client)+" buttons="+buttons(client)+" hand="+Registries.ITEM.getId(client.player.getMainHandStack().getItem())+" position="+client.player.getPos());
  }catch(Throwable failure){done=true;RESULT.addProperty("status","FAIL_ACTUAL_CLIENT_MENUS");RESULT.addProperty("failedStep",index<PLAN.size()?PLAN.get(index).name():"initialize");RESULT.addProperty("failureClass",failure.getClass().getName());RESULT.addProperty("failure",String.valueOf(failure.getMessage()));RESULT.addProperty("completedSteps",STEPS.size());client.options.attackKey.setPressed(false);}
  return done;
 }
 private static void initialize(MinecraftClient c)throws Exception{

  JsonObject input=JsonParser.parseString(Files.readString(Path.of(marker))).getAsJsonObject();
  if(!"dw-review-v10-client-input-v1".equals(text(input,"schema")))throw new IllegalStateException("Unexpected V10 client marker schema");
  JsonObject author;if(input.get("authorOutput").isJsonObject())author=input.getAsJsonObject("authorOutput");else author=JsonParser.parseString(Files.readString(Path.of(input.get("authorOutput").getAsString()))).getAsJsonObject();
  JsonElement raw=author.get("fixtures");JsonArray fixtures=raw!=null&&raw.isJsonArray()?raw.getAsJsonArray():author.getAsJsonArray("objects");if(fixtures==null)throw new IllegalStateException("Author fixture array absent");
  for(JsonElement e:fixtures){JsonObject o=e.getAsJsonObject();String key=text(o,"key");if(key.isBlank()||!o.has("uuid")||!o.has("root"))continue;JsonArray pos=o.getAsJsonArray("root");FIXTURES.put(key,new Fixture(key,text(o,"kind"),UUID.fromString(text(o,"uuid")),new BlockPos(pos.get(0).getAsInt(),pos.get(1).getAsInt(),pos.get(2).getAsInt())));}
  for(String required:List.of("lamp_A","lamp_B","lamp_C","lamp_D","lever_1","lever_2","lever_3","architecture_door","rp_door","wood_gate"))fixture(required);
  if(input.has("menus"))mode=text(input.getAsJsonObject("menus"),"mode");if(mode.isBlank())mode="AUTHOR";if(!Set.of("AUTHOR","REENTER").contains(mode))throw new IllegalStateException("Unsupported menus mode");
  RESULT.addProperty("schema","dw-review-v10-actual-client-menus-v1");RESULT.addProperty("status","RUNNING");RESULT.addProperty("mode",mode);RESULT.addProperty("artifactSha256",text(input,"artifactSha256"));RESULT.addProperty("manualVisualAcceptance","NOT_RUN");RESULT.addProperty("directMutationEndpointsUsedAsGuiProof",false);RESULT.addProperty("scope","Automated actual Screen.mouseClicked/charTyped/keyPressed, mouse input and ordinary interactionManager packets; not human acceptance");RESULT.add("steps",STEPS);RESULT.add("gestures",GESTURES);RESULT.add("screenshots",SCREENSHOTS);

  RESULT.addProperty("sourceWorld",c.getServer().getSavePath(net.minecraft.util.WorldSavePath.ROOT).toString());
  if(!c.player.isCreative())throw new IllegalStateException("First entry must already be Creative; no temporary mode fix in menus QA");

 }

  if(action==BuildingTool.Action.LAMP_SOURCE||action==BuildingTool.Action.LAMP_TARGET){JsonObject lamps=new Gson().toJsonTree(dev.dreamwalker.bloodbornerp.lamp.LampEditor.view(player,null)).getAsJsonObject();JsonObject node=object(lamps,"node");row.addProperty("actualSelectedLampUuid",text(node,"entityUuid"));if(action==BuildingTool.Action.LAMP_SOURCE)consequence=bool(lamps,"selected")&&text(node,"entityUuid").equals(f.uuid().toString());else{JsonObject wrapper=new JsonObject();wrapper.add("lamps",lamps);consequence=text(node,"entityUuid").equals(fixture("lamp_A").uuid().toString())&&connectionDirection(wrapper,fixture("lamp_A").uuid(),f.uuid(),true,true);row.addProperty("line",MAIN_LINE);row.addProperty("actualConnectionDefaultBidirectional",consequence);}}

 }
